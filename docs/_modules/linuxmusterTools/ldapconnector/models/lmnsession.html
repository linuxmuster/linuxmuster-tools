    def get_first_passwords(self):
        response = {}

        for member in self.members:
            lmnuser = lr.get(f'/users/{member}', dict=False)
            response[member] = {
                'firstPassword': lmnuser.sophomorixFirstPassword,
                'firstPasswordStillSet': lmnuser.test_first_password(),
            }
        return response

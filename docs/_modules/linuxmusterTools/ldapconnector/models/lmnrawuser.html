        self.printers = self.extract_printers(self.memberOf)
        self.dn = self.distinguishedName
        self.school = self.sophomorixSchoolname
        self.extract_management()
        self.parse_sessions()
        self.parse_exam()

            self.create_custom_fields_objects(custom_fields_config)

            self.customFields = {}
